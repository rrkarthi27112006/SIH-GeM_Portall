# Cell 25: Cross-Verify Scraped Bidder Data

def verify_bidder(row):
    issues = []

    if row["GSTIN"] != "VALID":
        issues.append("GSTIN issue")

    if row["PAN"] != "VALID":
        issues.append("PAN issue")

    if row["Udyam Status"] != "VALID":
        issues.append("Udyam issue")

    if row["Verification Status"] != "VERIFIED":
        issues.append("Portal verification issue")

    if len(issues) == 0:
        result = "MATCHED"
    else:
        result = "MISMATCH"

    return pd.Series({
        "Cross_Verification": result,
        "Issues_Detected": ", ".join(issues) if issues else "None"
    })


verification_results = scraped_df.apply(
    verify_bidder,
    axis=1
)

    [scraped_df, verification_results],
    axis=1
)

print("Cross-Verification Completed!\n")

def generate_compliance(row):

    if row["Cross_Verification"] == "MATCHED":
        status = "compliant"
        risk = "Low"
        recommendation = "All external verification details matched."

    elif "GSTIN issue" in row["Issues_Detected"] and \

        risk = "High"
        recommendation = "Critical identity verification mismatch detected."

    else:
        status = "minor_issue"
        risk = "Medium"
        recommendation = "Manual review required for detected verification issues."

    return pd.Series({
        "Compliance_Status": status,
        "Risk_Level": risk,
        "Recommendation": recommendation
    })

def predict_from_web_verification(row):

    # Convert scraped verification into model features
    gstin_valid = row["GSTIN"] == "VALID"
    pan_valid = row["PAN"] == "VALID"
    udyam_present = row["Udyam Status"] != "NOT FOUND"
    udyam_valid = row["Udyam Status"] == "VALID"

    # Create model input
    model_input = pd.DataFrame([{
        "tender_category": "IT & Computers",
        "tender_value_inr": 5000000,
        "turnover_required_inr": 3000000,
        "gstin_valid": gstin_valid,
        "pan_valid": pan_valid,
        "udyam_present": udyam_present,
        "udyam_valid": udyam_valid,
        "document_type_flagged": "None",
        "document_status": row["Verification Status"],
        "mismatch_type": (
            "Verification mismatch"
            if row["Cross_Verification"] == "MISMATCH"
            else "None"
        )
    }])

    # Boolean → integer
    for col in model_input.select_dtypes(include="bool").columns:
        model_input[col] = model_input[col].astype(int)

    # One-hot encoding
    model_input = pd.get_dummies(
        model_input,
        columns=cat_cols_clean,
        dummy_na=True
    )

    # Match training features
    model_input = model_input.reindex(
        columns=X_train_clean.columns,
        fill_value=0
    )

    # ML prediction
    prediction = rf_clean.predict(model_input)[0]

    return prediction

        "document_type_flagged": "None",
        "document_status": verification["verification_status"],
        "mismatch_type": (
            "Verification mismatch"
            if verification["verification_status"] != "VERIFIED"
            else "None"
        )
    }])

    # 4. Convert boolean values
    for col in model_input.select_dtypes(include="bool").columns:
        model_input[col] = model_input[col].astype(int)

    # 5. Encode categorical features
    model_input = pd.get_dummies(
        model_input,
        columns=cat_cols_clean,
        dummy_na=True
    )

    # 6. Match training features
    model_input = model_input.reindex(
        columns=X_train_clean.columns,
        fill_value=0
    )

    # 7. ML prediction
    prediction = rf_clean.predict(model_input)[0]

    probabilities = rf_clean.predict_proba(model_input)[0]
    confidence = probabilities.max() * 100

    return prediction, confidence
